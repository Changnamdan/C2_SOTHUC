let rawTeacherLogs = [];
let rawTeacherStats = [];
/**
* Tải dữ liệu báo cáo từ Google Apps Script
*/
async function fetchTeacherResults() {
const syncMsg = document.getElementById('td-sync-msg');
const scriptUrl = teacherConfig.scriptUrl;
if (!scriptUrl) {
alert("Vui lòng cấu hình URL Google Apps Script trong mục Cài đặt!");
return;
}
syncMsg.textContent = "⏳ Đang tải dữ liệu từ Google Sheets...";
try {
const res = await fetch(scriptUrl);
const data = await res.json();
rawTeacherLogs = data.logs || [];
rawTeacherStats = data.stats || [];
syncMsg.textContent = "✅ Đã cập nhật dữ liệu học sinh mới nhất!";
document.getElementById('td-last-update').textContent = "Lần cuối: "
+ new Date().toLocaleTimeString();
updateClassFilterOptions();
renderTeacherDashboard();
} catch (err) {
console.error(err);
syncMsg.textContent = " Không thể kết nối với Google Apps Script.
Hiển thị dữ liệu bộ nhớ đệm.";
renderTeacherDashboard();
}
}
/**
* Cập nhật danh sách các lớp trong bộ lọc Dropdown
*/
function updateClassFilterOptions() {
const select = document.getElementById('td-class-filter');
const classes = [...new Set(rawTeacherLogs.map(item =>
item.className))].filter(Boolean);
select.innerHTML = '<option value="ALL">Lớp: Tất cả các lớp</option>';
classes.forEach(c => {
select.innerHTML += `<option value="${c}">Lớp: ${c}</option>`;
});
}
/**
* Hiển thị Dashboard Giáo viên theo lớp được chọn
*/
function renderTeacherDashboard() {
const selectedClass = document.getElementById('td-class-filter').value;
// Lọc log bài làm theo lớp
const filteredLogs = (selectedClass === "ALL")
? rawTeacherLogs
: rawTeacherLogs.filter(item => item.className === selectedClass);
// 1. Render các thẻ KPI Tổng quan
renderKpiGrid(filteredLogs);
// 2. Render 4 Góc nhìn phân tích (Tabs)
renderSupportView(filteredLogs); // View 3: Dạng cả lớp sai nhiều nhất
& học sinh cần hỗ trợ
renderRankingView(filteredLogs); // View 1: Bảng xếp hạng
renderWeaknessView(filteredLogs); // View 2: Dạng yếu nhất từng người
renderChartView(selectedClass); // View 4: Biểu đồ tỉ lệ sai 5 dạng
bài
}
// --- VIEW 3: DẠNG CẢ LỚP SAI NHIỀU NHẤT + TÊN HỌC SINH CẦN HỖ TRỢ ---
function renderSupportView(logs) {
const tbody = document.getElementById('td-support-tbody');
tbody.innerHTML = "";
const topicMap = {};
CHAPTER2_TOPICS.forEach(t => topicMap[t] = []);
logs.forEach(log => {
if (log.wrongTopics && log.wrongTopics !== "Không có") {
const topicsArr = log.wrongTopics.split(", ");
topicsArr.forEach(topic => {
if (topicMap[topic] &&
!topicMap[topic].includes(log.studentName)) {
topicMap[topic].push(log.studentName);
}
});
}
});
CHAPTER2_TOPICS.forEach(topic => {
const students = topicMap[topic] || [];
let studentPills = students.length > 0
? students.map(s => `<span class="pill-
student">${s}</span>`).join("")
: `<span class="pill-good"> Tốt - Không có học sinh làm
sai</span>`;
tbody.innerHTML += `
<tr>
<td style="font-weight:800; color:#0f5c52;">${topic}</td>
<td>${studentPills}</td>
</tr>
`;
});
}
// --- VIEW 1: BẢNG XẾP HẠNG HỌC SINH ---
function renderRankingView(logs) {
const tbody = document.getElementById('td-ranking-tbody');
tbody.innerHTML = "";
// Sắp xếp theo điểm/tỷ lệ đúng giảm dần
const sorted = [...logs].sort((a, b) => parseInt(b.accuracy) -
parseInt(a.accuracy));
sorted.forEach((item, index) => {
tbody.innerHTML += `
<tr>
<td><b>#${index + 1}</b></td>
<td>${item.studentName}</td>
<td><span class="pill-student">${item.className}</span></td>
<td>${item.characterClass}</td>
<td>⭐ ${item.score * 2}</td>
<td><b>${item.score}/${item.totalQuestions}</b>
(${item.accuracy})</td>
</tr>
`;
});
}
// --- VIEW 2: DẠNG YẾU NHẤT CỦA TỪNG HỌC SINH ---
function renderWeaknessView(logs) {
const tbody = document.getElementById('td-weakness-tbody');
tbody.innerHTML = "";
logs.forEach(item => {
const weakTopic = item.wrongTopics !== "Không có" ?
item.wrongTopics.split(", ") : "Không";
const statusPill = weakTopic !== "Không"
? `<span class="pill-weak">${weakTopic}</span>`
: `<span class="pill-good">Thành thạo tốt</span>`;
tbody.innerHTML += `
<tr>
<td><b>${item.studentName}</b></td>
<td>${item.className}</td>
<td>${statusPill}</td>
<td>${100 - parseInt(item.accuracy)}% câu sai</td>
</tr>
`;
});
}
// --- VIEW 4: BIỂU ĐỒ TỈ LỆ LÀM SAI THEO DẠNG BÀI ---
function renderChartView(selectedClass) {
const container = document.getElementById('td-chart-container');
container.innerHTML = "";
rawTeacherStats.forEach(stat => {
const wrongPctVal = parseInt(stat.wrongPct) || 0;
container.innerHTML += `
<div class="chart-row">
<div class="chart-label">${stat.topic}</div>
<div class="chart-track">
<div class="chart-bar-bg">
<div class="chart-bar-val" style="width:
${wrongPctVal}%;"></div>
</div>
<span class="chart-pct">${wrongPctVal}%</span>
</div>
</div>
`;
});
}
